# Development probe scoped to the disposable shell's window, not the desktop.
# Run from ignored build/issue82. This does not claim screen-reader speech.
$ErrorActionPreference = 'Stop'
Add-Type -AssemblyName UIAutomationClient
Add-Type -AssemblyName UIAutomationTypes
$spikeOutputFiles = @('uia-local-stdout.txt', 'uia-local-stderr.txt', 'uia.json')
foreach ($spikeOutputFile in $spikeOutputFiles) {
    if (Test-Path -LiteralPath $spikeOutputFile) { Remove-Item -LiteralPath $spikeOutputFile -Force }
}
$spikePython = (Resolve-Path 'py310/Scripts/python.exe').Path
$spikeArguments = '/d /s /c ""{0}" shell.py --uia 1>uia-local-stdout.txt 2>uia-local-stderr.txt"' -f $spikePython
$spikeProcess = Start-Process -FilePath $env:ComSpec -ArgumentList $spikeArguments -WindowStyle Hidden -PassThru
$spikeWindow = [IntPtr]::Zero
for ($attempt = 0; $attempt -lt 80; $attempt++) {
    $spikeProcess.Refresh()
    if (Test-Path 'uia-local-stdout.txt') {
        foreach ($spikeLine in Get-Content 'uia-local-stdout.txt') {
            if ($spikeLine.StartsWith('{')) {
                $spikeEvent = $spikeLine | ConvertFrom-Json
                if ($spikeEvent.event -eq 'window') { $spikeWindow = [IntPtr]$spikeEvent.hwnd }
            }
        }
    }
    if ($spikeWindow -ne [IntPtr]::Zero -or $spikeProcess.HasExited) { break }
    Start-Sleep -Milliseconds 100
}
$spikeElements = @()
if ($spikeWindow -ne [IntPtr]::Zero) {
    $spikeRoot = [System.Windows.Automation.AutomationElement]::FromHandle($spikeWindow)
    $spikeChildren = $spikeRoot.FindAll([System.Windows.Automation.TreeScope]::Descendants, [System.Windows.Automation.Condition]::TrueCondition)
    foreach ($spikeChild in $spikeChildren) {
        $spikeCurrent = $spikeChild.Current
        $spikeElements += @{ name=$spikeCurrent.Name; role=$spikeCurrent.ControlType.ProgrammaticName; keyboard_focusable=$spikeCurrent.IsKeyboardFocusable; offscreen=$spikeCurrent.IsOffscreen }
    }
}
if (-not $spikeProcess.WaitForExit(30000)) {
    throw 'UIA process did not finish within the observation bound'
}
$spikeProcess.Refresh()
$spikeResult = @{
    window_found=($spikeWindow -ne [IntPtr]::Zero)
    elements=$spikeElements
    exit_code=$spikeProcess.ExitCode
    announcements_verified=$false
}
$spikeResult | ConvertTo-Json -Depth 5 | Set-Content -Encoding UTF8 'uia.json'
& $spikePython evidence_contract.py --uia-result uia.json --stderr uia-local-stderr.txt --stdout uia-local-stdout.txt
if ($LASTEXITCODE -ne 0) {
    throw 'UIA diagnostics failed; inspect local files'
}
if (-not $spikeResult.window_found -or @($spikeResult.elements).Count -eq 0 -or $spikeResult.exit_code -ne 0) {
    throw 'UIA probe did not produce a successful window-bound result; inspect local files'
}
